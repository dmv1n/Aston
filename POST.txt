// Проверяем статус-код
pm.test("Status code is 200", function () {
    pm.response.to.have.status(200);
});

// Проверяем, что в ответе есть поля data, json, headers, url
pm.test("Response has required fields", function () {
    var jsonData = pm.response.json();
    pm.expect(jsonData).to.have.property('data');
    pm.expect(jsonData).to.have.property('json');
    pm.expect(jsonData).to.have.property('headers');
    pm.expect(jsonData).to.have.property('url');
});

// Проверяем, что отправленные данные (name и age) совпадают с тем, что вернулось в json
pm.test("Sent data matches response", function () {
    var jsonData = pm.response.json();
    pm.expect(jsonData.json.name).to.eql('Test');
    pm.expect(jsonData.json.age).to.eql(30);
});
